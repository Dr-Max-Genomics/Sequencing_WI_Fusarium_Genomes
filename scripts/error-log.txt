#!/usr/bin/env python3
"""
dedup_by_readid_keep_longest.py

Purpose
-------
Fixes read-ID duplication introduced when a MinKNOW acquisition is
stopped/interrupted mid-run (e.g. stopped early for storage reasons,
or basecalling continuing in the background after the run stops).
In that situation the SAME read_id can be written into fastq_pass
more than once, with DIFFERENT sequence content/length each time
(an early, partially-flushed copy and a later, complete copy).

Content-based dedup (e.g. `seqkit rmdup -s`, the default mode) does
NOT catch these, because the two copies have different sequences.
`seqkit rmdup -n` (dedup by ID) catches them but keeps whichever
copy appears FIRST in the file -- which may be the short/partial
one if batches got concatenated out of completion order.

This script dedups by read ID and always keeps the LONGEST record
per ID (on the assumption that the complete read is the longer of
the two). It runs in two passes over the file so memory use stays
bounded by the number of unique read IDs, not by file size.

Usage
-----
    python3 dedup_by_readid_keep_longest.py input.fastq[.gz] output.fastq[.gz]

A short summary (records read, unique IDs, duplicated IDs, records
dropped) is written to stderr -- redirect it into your per-sample
log alongside the rest of the step's output.

Notes
-----
- Works on plain or gzip-compressed FASTQ (detected by .gz suffix).
- On a length tie between duplicate copies, the first one encountered
  in the file is kept (extremely rare in practice for this failure
  mode, since a partial flush is essentially never exactly as long
  as the completed read).
- Run this immediately after concatenating fastq_pass files for a
  barcode, BEFORE porechop -- adapter trimming can shift the two
  copies' lengths independently and make "keep longest" less reliable
  once it's been applied.
"""
import sys
import gzip


def opener(path, mode):
    if path.endswith(".gz"):
        return gzip.open(path, mode + "t")
    return open(path, mode)


def iter_fastq(path):
    with opener(path, "r") as fh:
        while True:
            header = fh.readline()
            if not header:
                return
            seq = fh.readline()
            plus = fh.readline()
            qual = fh.readline()
            if not qual:
                raise ValueError("Truncated FASTQ record near: " + header.strip())
            read_id = header[1:].split()[0].strip()
            yield read_id, header, seq, plus, qual


def main():
    if len(sys.argv) != 3:
        sys.exit(f"Usage: {sys.argv[0]} input.fastq[.gz] output.fastq[.gz]")

    in_path, out_path = sys.argv[1], sys.argv[2]

    # Pass 1: for every read ID, record how many times it occurs and
    # the longest sequence length seen for it.
    counts = {}
    max_len = {}
    total = 0
    for read_id, header, seq, plus, qual in iter_fastq(in_path):
        total += 1
        length = len(seq.rstrip("\n"))
        counts[read_id] = counts.get(read_id, 0) + 1
        if read_id not in max_len or length > max_len[read_id]:
            max_len[read_id] = length

    n_unique = len(counts)
    dup_ids = [rid for rid, c in counts.items() if c > 1]
    n_dup_records = sum(counts[rid] for rid in dup_ids)

    # Pass 2: write each ID's longest record exactly once.
    emitted = set()
    written = 0
    with opener(out_path, "w") as out:
        for read_id, header, seq, plus, qual in iter_fastq(in_path):
            if read_id in emitted:
                continue
            length = len(seq.rstrip("\n"))
            if length == max_len[read_id]:
                out.write(header)
                out.write(seq)
                out.write(plus)
                out.write(qual)
                emitted.add(read_id)
                written += 1

    sys.stderr.write(
        "[dedup_by_readid] input records:      {}\n".format(total) +
        "[dedup_by_readid] unique read IDs:    {}\n".format(n_unique) +
        "[dedup_by_readid] duplicated IDs:     {} ({} records involved)\n".format(
            len(dup_ids), n_dup_records) +
        "[dedup_by_readid] records written:    {}\n".format(written) +
        "[dedup_by_readid] records dropped:    {}\n".format(total - written)
    )

    # Dump the duplicate IDs themselves so you can spot-check a few
    # against the original per-batch fastq_pass files.
    if dup_ids:
        dup_list_path = out_path + ".duplicate_ids.txt"
        with open(dup_list_path, "w") as f:
            f.write("\n".join(sorted(dup_ids)) + "\n")
        sys.stderr.write(
            "[dedup_by_readid] duplicate ID list:  {}\n".format(dup_list_path)
        )


if __name__ == "__main__":
    main()
